% Combined IBR-adaptive 87L protection chain block diagram
% TRIP = [87L v 87LN v 87LN0 v TDCS] ^ gamma_gate
\begin{tikzpicture}[
  blk/.style={draw,fill=blue!8,rounded corners=3pt,minimum width=2.2cm,
              minimum height=0.65cm,font=\scriptsize,align=center},
  gate/.style={draw,fill=orange!12,rounded corners=3pt,minimum width=1.4cm,
               minimum height=0.65cm,font=\scriptsize,align=center},
  prep/.style={draw,fill=green!8,rounded corners=3pt,minimum width=2.2cm,
               minimum height=0.55cm,font=\scriptsize,align=center},
  orgate/.style={draw,fill=yellow!20,rounded corners=3pt,minimum width=1.2cm,
                 minimum height=0.55cm,font=\scriptsize,align=center},
  trip/.style={draw,fill=red!15,rounded corners=3pt,minimum width=1.2cm,
               minimum height=0.65cm,font=\scriptsize\bfseries,align=center},
  >=latex,node distance=0.5cm
]
% Input pre-processing column
\node[prep] (pifix)  at (0,3.0)  {Pi-section\\charging cancel};
\node[prep] (freqad) at (0,2.1)  {Freq-adaptive\\correction (TR-20)};
\node[prep] (hpre)   at (0,1.2)  {Harmonic\\pre-filter (TR-21)};
\node[prep] (seqex)  at (0,0.2)  {Sequence\\extraction};

% Protection elements column
\node[blk] (e87L)   at (3.6,3.2) {87L magnitude\\$|\Delta I_1| \geq I_{\rm thr}$};
\node[blk] (e87LN)  at (3.6,2.2) {87LN neg-seq\\$|\Delta I_2| \geq 0.06$\,pu};
\node[blk] (e87LN0) at (3.6,1.2) {87LN0 zero-seq\\$|\Delta I_0| \geq 0.04$\,pu};
\node[blk] (eTDCS)  at (3.6,0.2) {TDCS-87L transient\\$\Delta I_{\rm rms} \geq \Delta_{\rm thr}$};

% Arrows from pre-proc to elements
\foreach \s/\t in {pifix/e87L, freqad/e87L, hpre/e87L,
                   seqex/e87LN, seqex/e87LN0, freqad/eTDCS} {
  \draw[->,gray!60] (\s.east) -- (\t.west);
}

% Adaptive threshold label on 87L
\node[font=\tiny,gray!60,right=0pt of e87L,yshift=-8pt]
  {$I_{\rm thr}=\max(0.08,\,0.15 I_{\rm rest})$};

% OR gate
\node[orgate] (orgate) at (6.2,1.7) {OR};
\foreach \e in {e87L,e87LN,e87LN0,eTDCS} {
  \draw[->] (\e.east) -- (orgate.west|-\e) -- (orgate.west);
}

% Confidence gate
\node[gate] (gamma) at (7.9,1.7) {$\gamma$-gate\\$\gamma \geq 0.80$};
\draw[->] (orgate.east) -- (gamma.west);

% Trip output
\node[trip] (trip) at (9.6,1.7) {TRIP};
\draw[->,very thick,red!70!black] (gamma.east) -- (trip.west);

% Title labels
\node[font=\tiny\itshape,gray!70] at (0,3.75)   {Pre-processing};
\node[font=\tiny\itshape,gray!70] at (3.6,3.75) {Protection elements};
\node[font=\tiny\itshape,gray!70] at (7.9,3.75) {Trip logic};
\end{tikzpicture}
