  ylabel      = {Differential magnitude [pu]},
  ylabel style= {font=\small},
  xlabel      = {IBR current limit $k_{\rm ibr}$ [pu]},

  ({x}, {0.06});

% ---- Annotations for IBR-limited region ------------------------------------
