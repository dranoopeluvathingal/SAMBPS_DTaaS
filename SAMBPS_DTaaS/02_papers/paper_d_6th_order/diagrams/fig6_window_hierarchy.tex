\draw[thick, orange!80!black] (0, 0.95) -- (0.8, 0.95);

     {$L^\ast{\approx}150$\,ms\\validity boundary};

%% --- Tick marks on timeline ---------------------------------
\foreach \x/\lbl in {0/0, 2/100, 4/200, 6/300, 8/400, 10/500}{

  \node[below, font=\tiny] at (\x, -0.08) {\lbl};
}
