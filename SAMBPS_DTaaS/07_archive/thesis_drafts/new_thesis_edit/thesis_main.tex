%% ── Fix KOMA-Script 3.41 + microtype \MakeUppercase incompatibility ──────
%% LaTeX 2023 \MakeUppercase internally uses an optional-argument LaTeX3
%% implementation. When KOMA-Script writes this to the .toc as
%% \MakeUppercase []{text}, a microtype/LaTeX3-key interaction on the
%% second pass empties all TOC entries.
%% Fix: redefine \MakeUppercase as a plain 1-arg command via LaTeX3
%% \text_uppercase:n, so the .toc receives \MakeUppercase{text} (no []).
\microtypesetup{activate=false}
\ExplSyntaxOn
\DeclareRobustCommand \MakeUppercase [1] { \text_uppercase:n {#1} }
\ExplSyntaxOff
