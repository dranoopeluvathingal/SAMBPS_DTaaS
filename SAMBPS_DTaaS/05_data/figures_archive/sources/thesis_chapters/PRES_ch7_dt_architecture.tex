% ch7_dt_architecture.tex — Non-blocking digital twin architecture
\begin{tikzpicture}[
  every node/.style={font=\scriptsize},
  core/.style={draw=thesisblue, thick, fill=lightblue, rounded corners=4pt,
               minimum width=3.5cm, minimum height=1.0cm, align=center},
  mem/.style={draw=thesisgray, thick, fill=lightgray,
              minimum width=2.0cm, minimum height=0.65cm, align=center,
              font=\scriptsize},
]

%% CPU Core 1: EKF
\node[core, draw=thesisgreen, fill=lightgreen] (EKF) at (-3.5, 0)
  {\textbf{Core 1: EKF}\\Update cycle: 20~ms};

%% CPU Core 2: Relay Logic
\node[core] (RELAY) at (3.5, 0)
  {\textbf{Core 2: Relay Logic}\\Sample cycle: 50~$\mu$s};

%% Shadow register (shared memory)
\node[mem] (SR) at (0, 0) {Shadow\\Register\\$\hat{\kibr}, \hat{\alpha}, \hat{\Zline}$};

%% Write arrow from EKF
\draw[arr, thesisgreen, very thick]
  (EKF.east) -- node[above, font=\tiny]{Write (atomic)} (SR.west);

%% Read arrow to Relay
\draw[arr, thesisblue, very thick]
  (SR.east) -- node[above, font=\tiny]{Read (atomic)} (RELAY.west);

%% Inputs to EKF
\node[sensor] (CIA) at (-6.5, 1.5) {$\phasor{I}_A$\\local CT};
\node[sensor] (CIB) at (-6.5, -1.5) {$\phasor{I}_B$\\GOOSE};
\draw[dataarrow] (CIA.east) -- (EKF.west |- CIA);
\draw[dataarrow] (CIB.east) -- (EKF.west |- CIB);
\draw[dataarrow, dashed] (EKF.west) -- (EKF.west -| CIA);

%% Inputs to Relay Logic
\node[sensor] (VAB) at (6.5, 1.5) {$V_{abc}$\\local VT};
\node[sensor] (IAB) at (6.5, -1.5) {$\phasor{I}_A$\\relay CT};
\draw[dataarrow] (VAB.west) -- (RELAY.east |- VAB);
\draw[dataarrow] (IAB.west) -- (RELAY.east |- IAB);

%% Output from Relay
\node[block, draw=thesisred, fill=red!10, minimum width=1.5cm, minimum height=0.6cm]
  (TRIP) at (3.5, -2.5) {TRIP\\$\leq 5$~ms};
\draw[arr, thesisred, very thick] (RELAY.south) -- (TRIP.north);

%% IEEE 1588 PTP clock
\node[block, fill=yellow!20, draw=thesisyellow, minimum width=2cm, minimum height=0.5cm,
      font=\tiny] (PTP) at (0, -2.5) {IEEE~1588 PTP\\Clock ($\pm 1\,\mu$s)};
\draw[dataarrow] (PTP.west) -- ++(-2.5,0) |- (EKF.south);
\draw[dataarrow] (PTP.east) -- ++(2.5,0) |- (RELAY.south);

%% Non-blocking guarantee annotation
\node[draw=thesisblue!50, rounded corners=2pt, fill=lightblue!30,
      font=\tiny, align=center, text width=3cm] at (0,-3.8)
  {Non-blocking guarantee:\\TRIP never waits\\for EKF update};

\end{tikzpicture}
