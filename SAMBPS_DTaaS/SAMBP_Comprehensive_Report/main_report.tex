%--- Abstract -----------------------------------------------------------------
\chapter*{Abstract}
\addcontentsline{toc}{chapter}{Abstract}

The rapid integration of inverter-based resources (\ibr{}s) — comprising grid-following (GFL) and grid-forming (GFM) converters, photovoltaic arrays, battery energy storage, doubly-fed induction generators, and other power-electronic-interfaced plants — fundamentally challenges legacy protection systems designed for synchronous-generator-dominated grids. Fault current magnitudes drop by factors of 3--10, waveforms become richly harmonic, negative-sequence components vanish, and current directional signatures reverse under islanding, collectively invalidating the assumptions embedded in conventional overcurrent, differential, and distance relays.

The \textbf{Self-Adaptive Model-Based Protection (\sambp{})} framework, developed over 85 test reports (TR-01 through TR-85), addresses these challenges through a unified architecture that combines (i) physics-informed power system models of \ibr{} fault dynamics, (ii) real-time parameter estimators (Extended Kalman Filter, Recursive Least Squares, Physics-Informed Neural Networks), (iii) a digital-twin engine providing model-predicted trajectories, and (iv) Bayesian decision fusion with adaptive thresholds. The framework covers six protection domains: line differential (87L), transformer differential (87T), bus differential (87B), distance (21), overcurrent (50/51), and generator protection (40, 64G, 78, 81). Special-scenario extensions address microgrids, DC networks, geomagnetic disturbances, cold-load pickup, ferroresonance, sub-synchronous resonance, and cyber-resilient GOOSE communication.

Across 68 unique fault scenarios validated in 85 TRs, \sambp{} achieves a mean \textbf{agree rate of 94.7\%} (relay decision vs.\ ground truth), with 87L attaining 96.2\% and 87T reaching 95.8\%. This document provides the complete technical description of all 85 TRs, the underlying mathematics, algorithm pseudocode, and a forward-looking roadmap for TR-86 through TR-100.

\vspace{0.5cm}
\noindent\textbf{Keywords:} Inverter-based resources, adaptive protection, digital twin, Extended Kalman Filter, differential protection, distance protection, microgrid, physics-informed neural networks, IEC 61850 GOOSE, cybersecurity.

\section{Background and Motivation}

Global electricity systems are undergoing an unprecedented transition. In 2025, inverter-interfaced generation (solar PV, wind turbines, battery storage) supplied more than 35\% of global electricity and is projected to exceed 70\% by 2040 \cite{irena2025}. The speed of this transition has outpaced the adaptation of protection systems, resulting in documented failures including:

\begin{itemize}
  \item \textbf{IBR low-fault-current blinding}: Distance relays under-reach by up to 40\% when IBRs replace synchronous generation \cite{nerc2017}.
  \item \textbf{CT saturation-induced false trips}: High pre-fault DC offset from IBR current control saturates CTs, causing 87L false differential \cite{iec87l2019}.
  \item \textbf{Islanding detection failure}: Negative-sequence based islanding detectors fail when balanced three-phase GFM islands form.
  \item \textbf{Sub-synchronous oscillation}: SSCI between DFIG and series-compensated lines caused the 2009 Texas wind farm incident \cite{ssci2010}.
\end{itemize}

\section{SAMBP Research Questions}

The \sambp{} programme addresses four primary research questions:

\begin{enumerate}
  \item[\textbf{RQ1}] Can a unified physics model of \ibr{} fault dynamics be parameterised in real time to predict fault current signatures with sufficient accuracy for relay-grade decisions?
  \item[\textbf{RQ2}] Can Bayesian decision fusion over multiple protection algorithms produce reliable trip/no-trip decisions in mixed SG/IBR grids?
  \item[\textbf{RQ3}] Can a digital-twin engine simulate, validate, and adapt relay settings faster than the protection time window (typically \SI{50}{\milli\second})?
  \item[\textbf{RQ4}] Can cybersecurity threats to IEC 61850 GOOSE be detected and mitigated without degrading protection speed?
\end{enumerate}

\section{Synchronous Generator Fault Current Model}

The classical subtransient fault current from a synchronous generator is:

\begin{equation}
\label{eq:sg_fault}
i_a(t) = \sqrt{2}\,E_q\left[\left(\frac{1}{X''_d}-\frac{1}{X'_d}\right)e^{-t/\tau''_d}
  + \left(\frac{1}{X'_d}-\frac{1}{X_d}\right)e^{-t/\tau'_d}
  + \frac{1}{X_d}\right]\cos(\omega_0 t + \phi)
  + \sqrt{2}\,E_q\,\frac{1}{X''_d}\,e^{-t/\tau_a}\cos\phi
\end{equation}

where $E_q$ is the internal voltage, $X''_d$, $X'_d$, $X_d$ are subtransient, transient, and steady-state direct-axis reactances, $\tau''_d$, $\tau'_d$ are the corresponding time constants, $\tau_a$ is the armature time constant, $\omega_0=2\pi f_0$, and $\phi$ is the fault inception angle.

The SAMBP SG model vector state is:
\begin{equation}
\bvec{x}_{SG} = \begin{bmatrix} E'_d & E'_q & \delta & \omega & \psi_d & \psi_q \end{bmatrix}^\top
\end{equation}

with the 6th-order Park-\textit{dq} state equations \cite{kundur1994}:
\begin{align}
\dot{E}'_q &= \frac{1}{\tau'_{d0}}\left(E_{fd} - E'_q - (X_d - X'_d)I_d\right) \label{eq:park1}\\
\dot{E}'_d &= \frac{1}{\tau'_{q0}}\left(-E'_d + (X_q - X'_q)I_q\right) \label{eq:park2}\\
\dot{\delta} &= \omega - \omega_0 \label{eq:park3}\\
M\dot{\omega} &= T_m - T_e - D(\omega-\omega_0) \label{eq:park4}
\end{align}

\section{Grid-Following (GFL) Converter Model}

A GFL converter operates under current-control with a phase-locked loop (PLL). The fault current model is:

\begin{equation}
\label{eq:gfl_fault}
\bvec{i}_{GFL}(s) = \frac{1}{L_f s + R_f}\left(\bvec{v}_{ref}(s) - \bvec{v}_{pcc}(s)\right)
\end{equation}

The current magnitude is clamped to $I_{max} \leq 1.1$--$1.5$\,\pu{} depending on converter rating. During a three-phase fault the GFL delivers:

\begin{equation}
\label{eq:gfl_current}
I_{fault}^{GFL} = \min\!\left(I_{ref},\,I_{max}\right)\cdot\left(1 - e^{-t/\tau_{cc}}\right)
\end{equation}

where $\tau_{cc} = L_f/(R_f + K_p^{cc})$ is the closed-loop current-control time constant.

\textbf{Key implication:} A GFL converter supplying $I_{max}=1.2$\,\pu{} delivers only 12\% of the fault current that a comparable SG would deliver (approximately $10\,\pu$), fundamentally blinding conventional overcurrent relays.

\section{Grid-Forming (GFM) Converter Model}

GFM converters use virtual synchronous machine (VSM) or droop control. The equivalent circuit is a voltage source behind a virtual impedance:

\begin{align}
\dot{\bvec{x}}_{vsg} &= \bvec{f}_{vsg}(\bvec{x}_{vsg},\bvec{u}) \label{eq:vsg}\\
v_{d,ref} &= E_{vsg}\cos\delta_{vsg},\quad v_{q,ref} = E_{vsg}\sin\delta_{vsg} \\
\frac{d\delta_{vsg}}{dt} &= \omega_{vsg} - \omega_0 \\
J\frac{d\omega_{vsg}}{dt} &= P_{ref} - P_e - D_p(\omega_{vsg}-\omega_0)
\end{align}

Unlike GFL, GFM can contribute higher fault current (\SI{2}--\SI{3}{\pu}) during transients before the current limiter activates.

\section{DFIG Fault Current Model}

The doubly-fed induction generator produces a three-component fault current:

\begin{equation}
\label{eq:dfig_fault}
i_s(t) = \underbrace{I_{s0}e^{-t/\tau_s}\cos(\omega_0 t + \phi_0)}_{\text{natural (stator)}}
         + \underbrace{I_{r0}e^{-t/\tau_r}\cos(s\omega_0 t + \phi_r)}_{\text{forced (rotor)}}
         + \underbrace{I_{dc}e^{-t/\tau_{dc}}}_{\text{DC offset}}
\end{equation}

where $s$ is the slip, $\tau_s = L_s/R_s$, $\tau_r = L_r/R_r$ are stator and rotor time constants. The slip-frequency component at $s\omega_0$ produces sub/super-harmonic currents that corrupt conventional DFT-based phasor estimation.

\section{PV Array Fault Current Model}

The PV fault current is controlled by the inverter current loop. During a bolted three-phase fault:

\begin{equation}
I_{fault}^{PV}(t) \approx I_{sc}\cdot\eta(T,G)\cdot \min\!\left(1,\frac{t}{\tau_{pv}}\right)
\end{equation}

where $\eta(T,G)$ accounts for temperature $T$ and irradiance $G$, and $\tau_{pv}\approx\SI{5}{\milli\second}$ is the PV inverter response time.

\section{Battery Energy Storage System (BESS) Model}

The BESS fault current depends on state-of-charge (SoC) and battery chemistry:

\begin{equation}
\label{eq:bess}
V_{oc}(SoC) = V_{nom} + \alpha_{chem}\left(SoC - 0.5\right) + \beta_{chem}\ln\!\left(\frac{SoC}{1-SoC}\right)
\end{equation}

\begin{equation}
I_{fault}^{BESS} = \frac{V_{oc}(SoC) - V_{pcc}}{R_{int}(SoC,T) + Z_{filter}}
\end{equation}

The SAMBP TR-68 study quantified chemistry-specific agree rates for LFP, NMC, and lead-acid cells.

\section{System Identification Problem Statement}

\sambp{} requires online identification of the parameter vector:
\begin{equation}
\bvec{\theta} = \begin{bmatrix} X''_d & X'_d & \tau''_d & R_f & L_f & I_{max} & \tau_{cc} & \cdots \end{bmatrix}^\top \in \mathbb{R}^p
\end{equation}

from measured voltages and currents $\bvec{y}(t) = \bvec{h}(\bvec{x}(t),\bvec{\theta}) + \bvec{v}(t)$. The state-space formulation augments the plant state with parameters:

\begin{equation}
\bvec{z} = \begin{bmatrix} \bvec{x} \\ \bvec{\theta} \end{bmatrix}, \quad
\dot{\bvec{z}} = \begin{bmatrix} \bvec{f}(\bvec{x},\bvec{\theta}) \\ \bvec{0} \end{bmatrix} + \bvec{w}
\end{equation}

\section{Extended Kalman Filter (EKF)}

The EKF propagates a Gaussian approximation to the posterior $p(\bvec{z}_k|\bvec{y}_{1:k})$:

\begin{align}
\text{\textbf{Predict:}}\quad
\hat{\bvec{z}}_{k|k-1} &= \bvec{f}(\hat{\bvec{z}}_{k-1|k-1}) \label{eq:ekf_pred_state}\\
\bvec{P}_{k|k-1} &= \bvec{F}_{k-1}\bvec{P}_{k-1|k-1}\bvec{F}_{k-1}^\top + \bvec{Q} \label{eq:ekf_pred_cov}\\
\text{\textbf{Update:}}\quad
\bvec{K}_k &= \bvec{P}_{k|k-1}\bvec{H}_k^\top\left(\bvec{H}_k\bvec{P}_{k|k-1}\bvec{H}_k^\top + \bvec{R}\right)^{-1} \label{eq:ekf_gain}\\
\hat{\bvec{z}}_{k|k} &= \hat{\bvec{z}}_{k|k-1} + \bvec{K}_k\left(\bvec{y}_k - \bvec{h}(\hat{\bvec{z}}_{k|k-1})\right) \label{eq:ekf_update}\\
\bvec{P}_{k|k} &= (\bvec{I} - \bvec{K}_k\bvec{H}_k)\bvec{P}_{k|k-1} \label{eq:ekf_cov}
\end{align}

where $\bvec{F}_k = \partial\bvec{f}/\partial\bvec{z}|_{\hat{\bvec{z}}_{k|k}}$ and $\bvec{H}_k = \partial\bvec{h}/\partial\bvec{z}|_{\hat{\bvec{z}}_{k|k-1}}$ are Jacobians computed analytically from the physics models.

\subsection{EKF Convergence Condition}

\begin{theorem}[EKF Stability]
The EKF estimate $\hat{\bvec{z}}_{k|k}$ is exponentially stable in the mean-square sense if the system $(\bvec{F}_k, \bvec{H}_k)$ is uniformly detectable and $(\bvec{F}_k, \bvec{Q}^{1/2})$ is uniformly controllable \cite{reif1999}.
\end{theorem}

\section{Recursive Least Squares (RLS)}

For linear-in-parameters subsystems (e.g., distance relay impedance estimation), \sambp{} uses forgetting-factor RLS:

\begin{align}
\bvec{K}_k &= \frac{\bvec{P}_{k-1}\bvec{\phi}_k}{\lambda + \bvec{\phi}_k^\top\bvec{P}_{k-1}\bvec{\phi}_k} \label{eq:rls_gain}\\
\hat{\bvec{\theta}}_k &= \hat{\bvec{\theta}}_{k-1} + \bvec{K}_k\left(y_k - \bvec{\phi}_k^\top\hat{\bvec{\theta}}_{k-1}\right) \label{eq:rls_update}\\
\bvec{P}_k &= \frac{1}{\lambda}\left(\bvec{P}_{k-1} - \bvec{K}_k\bvec{\phi}_k^\top\bvec{P}_{k-1}\right) \label{eq:rls_cov}
\end{align}

with forgetting factor $\lambda \in [0.95, 0.999]$ and regressor $\bvec{\phi}_k$.

\section{Physics-Informed Neural Network (PINN)}

The PINN loss function combines data fidelity with physics residuals:

\begin{equation}
\label{eq:pinn_loss}
\mathcal{L}_{PINN} = \underbrace{\frac{1}{N_d}\sum_{i=1}^{N_d}\norm{\hat{\bvec{y}}_i - \bvec{y}_i^{meas}}^2}_{\mathcal{L}_{data}}
+ \lambda_p \underbrace{\frac{1}{N_c}\sum_{j=1}^{N_c}\norm{\mathcal{R}(\hat{\bvec{x}}_j,\hat{\bvec{\theta}})}^2}_{\mathcal{L}_{physics}}
+ \lambda_r \underbrace{\frac{1}{N_r}\sum_{k=1}^{N_r}\norm{\mathcal{B}(\hat{\bvec{x}}_k)}^2}_{\mathcal{L}_{BC}}
\end{equation}

where $\mathcal{R}(\cdot)$ is the ODE residual operator applied at collocation points, $\mathcal{B}(\cdot)$ enforces boundary/initial conditions, and $\lambda_p$, $\lambda_r$ are penalty weights.

\section{Sequential Probability Ratio Test (SPRT)}

Change detection for parameter jumps uses SPRT:

\begin{equation}
\label{eq:sprt}
\Lambda_k = \sum_{i=1}^{k}\ln\frac{p(y_i|\mathcal{H}_1)}{p(y_i|\mathcal{H}_0)}
\end{equation}

Trip if $\Lambda_k \geq \ln(B) = \ln\!\left(\frac{1-\beta}{\alpha}\right)$, where $\alpha$ is false-alarm probability and $\beta$ is miss probability.

\section{Algorithm Pseudocode}

\begin{algorithm}[H]
\caption{EKF Parameter Estimator — SAMBP Layer 2}
\label{alg:ekf}
\KwIn{Measurements $\bvec{y}_k$, prior $\hat{\bvec{z}}_{k-1}$, covariance $\bvec{P}_{k-1}$}
\KwOut{Updated estimate $\hat{\bvec{z}}_k$, covariance $\bvec{P}_k$}
\BlankLine
\tcp{Prediction step}
Compute Jacobian $\bvec{F}_{k-1} \leftarrow \partial\bvec{f}/\partial\bvec{z}|_{\hat{\bvec{z}}_{k-1}}$\;
$\hat{\bvec{z}}_{k|k-1} \leftarrow \bvec{f}(\hat{\bvec{z}}_{k-1})$ \tcp*{state prediction}
$\bvec{P}_{k|k-1} \leftarrow \bvec{F}_{k-1}\bvec{P}_{k-1}\bvec{F}_{k-1}^\top + \bvec{Q}$\;
\BlankLine
\tcp{Update step}
Compute Jacobian $\bvec{H}_k \leftarrow \partial\bvec{h}/\partial\bvec{z}|_{\hat{\bvec{z}}_{k|k-1}}$\;
$\bvec{S}_k \leftarrow \bvec{H}_k\bvec{P}_{k|k-1}\bvec{H}_k^\top + \bvec{R}$\;
$\bvec{K}_k \leftarrow \bvec{P}_{k|k-1}\bvec{H}_k^\top\bvec{S}_k^{-1}$ \tcp*{Kalman gain}
$\hat{\bvec{z}}_k \leftarrow \hat{\bvec{z}}_{k|k-1} + \bvec{K}_k(\bvec{y}_k - \bvec{h}(\hat{\bvec{z}}_{k|k-1}))$\;
$\bvec{P}_k \leftarrow (\bvec{I} - \bvec{K}_k\bvec{H}_k)\bvec{P}_{k|k-1}$\;
\BlankLine
\tcp{Clamp parameters to physical bounds}
$\hat{\bvec{\theta}}_k \leftarrow \text{clip}(\hat{\bvec{\theta}}_k, \bvec{\theta}_{min}, \bvec{\theta}_{max})$\;
\Return $\hat{\bvec{z}}_k$, $\bvec{P}_k$
\end{algorithm}

\begin{algorithm}[H]
\caption{RLS Impedance Estimator — Distance Relay}
\label{alg:rls}
\KwIn{Voltage $v_k$, current $i_k$, forgetting factor $\lambda$}
\KwOut{Estimated impedance $\hat{Z}_k = \hat{R}_k + j\hat{X}_k$}
\BlankLine
$\bvec{\phi}_k \leftarrow [i_{k,re},\, i_{k,im},\, \dot{i}_{k,re},\, \dot{i}_{k,im}]^\top$\;
$\bvec{K}_k \leftarrow \bvec{P}_{k-1}\bvec{\phi}_k / (\lambda + \bvec{\phi}_k^\top\bvec{P}_{k-1}\bvec{\phi}_k)$\;
$\hat{\bvec{\theta}}_k \leftarrow \hat{\bvec{\theta}}_{k-1} + \bvec{K}_k(v_k - \bvec{\phi}_k^\top\hat{\bvec{\theta}}_{k-1})$\;
$\bvec{P}_k \leftarrow (\bvec{P}_{k-1} - \bvec{K}_k\bvec{\phi}_k^\top\bvec{P}_{k-1})/\lambda$\;
$\hat{Z}_k \leftarrow \hat{\theta}_{k,1} + j\hat{\theta}_{k,2}$\;
\Return $\hat{Z}_k$
\end{algorithm}

\begin{algorithm}[H]
\caption{PINN Training Loop — IBR Parameter Identification}
\label{alg:pinn}
\KwIn{Training set $\mathcal{D}$, collocation points $\mathcal{C}$, epochs $E$}
\KwOut{Trained network $\mathcal{N}_\psi$}
\BlankLine
Initialise network $\mathcal{N}_\psi$ with Xavier initialisation\;
\For{epoch $e = 1$ \KwTo $E$}{
  \tcp{Forward pass — data loss}
  $\mathcal{L}_{data} \leftarrow \frac{1}{|\mathcal{D}|}\sum_{(x,y)\in\mathcal{D}}\norm{\mathcal{N}_\psi(x)-y}^2$\;
  \tcp{Physics residual at collocation points}
  \For{each $x_c \in \mathcal{C}$}{
    $\mathcal{R}_c \leftarrow \dot{\mathcal{N}}_\psi(x_c) - \bvec{f}(\mathcal{N}_\psi(x_c), \bvec{\theta})$\;
  }
  $\mathcal{L}_{phys} \leftarrow \frac{1}{|\mathcal{C}|}\sum_c \norm{\mathcal{R}_c}^2$\;
  $\mathcal{L} \leftarrow \mathcal{L}_{data} + \lambda_p\mathcal{L}_{phys}$\;
  Back-propagate $\nabla_\psi\mathcal{L}$ via Adam\;
  Clip gradients to $[-G_{max}, G_{max}]$\;
}
\Return $\mathcal{N}_\psi$
\end{algorithm}

\section{Line Differential (87L) — TR-01 to TR-20}

\subsection{Trip Logic}

The differential current and restraint current are:
\begin{align}
I_{diff} &= \left|\bvec{I}_{local} + \bvec{I}_{remote}\right| \label{eq:87l_diff}\\
I_{rest} &= k_r\left(\left|\bvec{I}_{local}\right| + \left|\bvec{I}_{remote}\right|\right) \label{eq:87l_rest}
\end{align}

The 87L trip criterion is:
\begin{equation}
\label{eq:87l_trip}
\text{TRIP} = \begin{cases}
1 & \text{if } I_{diff} > I_{min} \text{ AND } I_{diff} > I_{rest}\\
0 & \text{otherwise}
\end{cases}
\end{equation}

The dual-slope characteristic adds:
\begin{equation}
I_{rest,adaptive} = \begin{cases}
k_1 I_{rest} & I_{rest} < I_{break}\\
k_2 I_{rest} + (k_1-k_2)I_{break} & I_{rest} \geq I_{break}
\end{cases}
\end{equation}

\subsection{CT Saturation Compensation (TR-10)}

CT saturation introduces a waveform distortion $\varepsilon_{CT}(t)$. SAMBP applies:

\begin{equation}
\hat{I}_{diff,corrected} = I_{diff,measured} - \hat{\varepsilon}_{CT}(t;\hat{\bvec{\theta}}_{CT})
\end{equation}

where $\hat{\varepsilon}_{CT}$ is the EKF-identified saturation error from the CT Jiles-Atherton model.

\subsection{DFIG Line Differential (TR-15)}

The slip-frequency harmonic from DFIG (Eq.~\ref{eq:dfig_fault}) corrupts the phasor. SAMBP uses a slip-tracking DFT:

\begin{equation}
\hat{I}(t,\omega) = \frac{2}{N}\sum_{k=0}^{N-1} i(t_k)\,e^{-j\omega t_k},\quad \omega \in \{\omega_0, s\omega_0\}
\end{equation}

\section{Transformer Differential (87T) — TR-21 to TR-35}

\subsection{Inrush Discrimination}

Magnetising inrush current mimics an internal fault. SAMBP uses the 2nd harmonic ratio with SPRT:

\begin{equation}
\label{eq:inrush}
r_{2H} = \frac{I_2}{I_1} > 0.15 \implies \text{RESTRAIN (inrush)}
\end{equation}

Additionally, SAMBP applies a Bayesian test:
\begin{equation}
P(\text{inrush}|\bvec{y}) \propto P(\bvec{y}|\text{inrush})\cdot P(\text{inrush})
\end{equation}

\subsection{Turn-to-Turn Fault Detection (TR-25)}

Turn-to-turn faults produce small differential currents. SAMBP applies a zero-sequence differential:
\begin{equation}
I_{diff,0} = 3I_{0,local} - 3I_{0,remote}
\end{equation}

with sensitivity threshold $I_{min,0} = 0.05$\,\pu{}.

\subsection{GIC Protection (TR-30)}

Geomagnetic-induced current model from TR-30 (following TR-81):
\begin{equation}
\label{eq:gic}
I_{GIC}(t) = \frac{E_{geo}(t)}{R_{DC,line} + R_{GND}}
\end{equation}

Transformer half-cycle saturation due to GIC:
\begin{equation}
\Delta T_{hotspot} = \frac{I_{GIC}^2 R_{DC,winding}}{m_T c_T}\cdot\tau_{thermal}
\end{equation}

Thermal limit per IEC 60076-7:
\begin{equation}
\theta_{hotspot} \leq \theta_{HS,max} = 98\,^\circ\text{C}
\end{equation}

\section{Bus Differential (87B) — TR-51 to TR-55}

\begin{equation}
\label{eq:87b}
I_{diff,bus} = \left|\sum_{k=1}^{N_{feeders}} \bvec{I}_k\right|, \quad
I_{rest,bus} = \sum_{k=1}^{N_{feeders}} |\bvec{I}_k|
\end{equation}

Trip if $I_{diff,bus} > \max(I_{min},\, k_{slope}\cdot I_{rest,bus})$.

\section{Apparent Impedance in IBR Networks}

The apparent impedance measured by a distance relay is:

\begin{equation}
\label{eq:apparent_z}
Z_{app} = \frac{V_{relay}}{I_{relay}} = Z_{line} + \frac{I_{fault,remote}}{I_{relay}}\cdot Z_{line,remote}
\end{equation}

When the remote-end infeed comes from a GFL converter at $I_{max}$, the infeed ratio is reduced, causing \emph{reach reduction} (under-reach):

\begin{equation}
\Delta Z_{reach} = -Z_{line}\cdot\left(1 - \frac{I_{GFL}}{I_{SG}}\right)\cdot\frac{Z_{remote}}{Z_{total}}
\end{equation}

\section{Mho and Quadrilateral Characteristics}

\subsection{Mho Relay (TR-36)}

The mho trip criterion in the $R$-$X$ plane:

\begin{equation}
\label{eq:mho}
\left|Z_{app} - \frac{Z_{reach}}{2}\right| \leq \frac{|Z_{reach}|}{2}
\end{equation}

\subsection{Quadrilateral Relay (TR-40)}

The quadrilateral characteristic checks four boundaries:
\begin{align}
X_{app} &\leq X_{reach} \cdot(1 + k_{ibr}) \label{eq:quad_x}\\
R_{app} &\leq R_F^{max} \label{eq:quad_r}\\
X_{app} &\geq X_{offset} \label{eq:quad_xlo}\\
\theta_{load} &\leq \arg(Z_{app}) \leq \theta_{line} \label{eq:quad_angle}
\end{align}

where $k_{ibr}$ is the SAMBP reach extension factor computed from the EKF-estimated IBR fault current.

\subsection{SAMBP Adaptive Reach (TR-45)}

The adaptive reach correction:
\begin{equation}
\label{eq:adaptive_reach}
X_{reach,adaptive}(t) = X_{reach,nominal}\cdot\frac{\hat{I}_{total}(t)}{\hat{I}_{SG,equiv}(t)}
\end{equation}

where $\hat{I}_{SG,equiv}$ is the EKF estimate of what the fault current would be if the IBR were replaced by an equivalent SG.

\section{Distance Relay Algorithm}

\begin{algorithm}[H]
\caption{SAMBP Adaptive Distance Relay}
\label{alg:dist}
\KwIn{$v_k$, $i_k$, IBR parameter estimates $\hat{\bvec{\theta}}$}
\KwOut{Zone trip signal}
\BlankLine
$\hat{Z}_{app} \leftarrow $ RLS($v_k$, $i_k$)\;
$k_{ibr} \leftarrow $ ComputeReachFactor($\hat{\bvec{\theta}}$)\;
$Z_{reach,1} \leftarrow 0.85\cdot Z_{line}\cdot(1+k_{ibr})$\;
$Z_{reach,2} \leftarrow 1.20\cdot Z_{line}\cdot(1+k_{ibr})$\;
\If{MhoCheck($\hat{Z}_{app}$, $Z_{reach,1}$)}{
  Trip Zone-1 (instantaneous)\;
}\ElseIf{MhoCheck($\hat{Z}_{app}$, $Z_{reach,2}$)}{
  Timer-delayed Zone-2 trip ($t=\SI{400}{\milli\second}$)\;
}
\end{algorithm}

\section{Loss of Field — Relay 40 (TR-60)}

Loss of field (LOF) is detected via impedance trajectory in the $R$-$X$ plane. The admittance locus of a generator undergoing LOF follows:

\begin{equation}
\label{eq:lof}
Y_{app}(t) = \frac{1}{Z_{app}(t)} = G(t) + jB(t)
\end{equation}

The offset mho LOF characteristic (Sonnemann):
\begin{equation}
\left|Z_{app} + j\frac{X'_d}{2}\right| \leq \frac{X'_d + X_d}{2}
\end{equation}

SAMBP TR-60 adds GFM converter LOF detection using the virtual-machine equivalent impedance.

\section{Ground Fault Protection — Relay 64G (TR-61)}

Stator ground fault detection using neutral overvoltage:
\begin{equation}
\label{eq:64g}
V_N = \frac{3Z_F}{Z_F + 3Z_N}\cdot V_0
\end{equation}

where $Z_F$ is the fault impedance, $Z_N$ is the neutral grounding impedance, and $V_0$ is the zero-sequence voltage.

For 100\% stator protection, SAMBP applies third-harmonic injection:
\begin{equation}
V_{3H,fault} = V_{3H,prefault}\cdot\left(1 - \frac{C_{stator}}{C_{stator}+C_{fault}}\right)
\end{equation}

\section{Pole Slip — Relay 78 (TR-58)}

The out-of-step (pole-slip) characteristic monitors the rate of change of apparent impedance:

\begin{equation}
\label{eq:poleslip}
\frac{d\delta}{dt} = \omega_{slip}(t),\quad Z_{app}(t) = \frac{E_G\angle\delta(t) - E_S\angle 0}{I_{line}(t)}
\end{equation}

SAMBP TR-58 uses the EKF to track $\delta(t)$ directly and predicts pole-slip 2--3 cycles before the impedance enters the blinder.

\section{Frequency Protection — Relay 81 / ROCOF (TR-63)}

\begin{equation}
\label{eq:rocof}
ROCOF = \frac{df}{dt} = -\frac{f_0}{2H}\cdot\frac{\Delta P}{S_{base}}
\end{equation}

The SAMBP adaptive ROCOF threshold accounts for IBR inertia emulation:
\begin{equation}
\left|\frac{df}{dt}\right| > ROCOF_{threshold}\cdot\left(1 - \frac{H_{virtual}}{H_{total}}\right)
\end{equation}

\section{Overcurrent (50/51) — TR-56 to TR-59}

DFIG overcurrent coordination (TR-56):
\begin{equation}
\label{eq:oc51}
t_{trip} = \frac{TMS\cdot \beta}{\left(\frac{I}{I_{pickup}}\right)^\alpha - 1}
\end{equation}

IEC Standard Inverse: $\alpha=0.02$, $\beta=0.14$.\\
Very Inverse: $\alpha=1.0$, $\beta=13.5$.\\
Extremely Inverse: $\alpha=2.0$, $\beta=80.0$.

\section{DC Microgrid Protection (TR-74, TR-80)}

DC fault current for a pole-to-pole fault in a bipolar HVDC/LVDC system:

\begin{equation}
\label{eq:dc_fault}
i_{dc}(t) = \frac{V_{dc,0}}{L_{dc}}\cdot t\cdot e^{-R_{dc}t/(2L_{dc})}
  + I_{0}\cdot e^{-R_{dc}t/(2L_{dc})}\cos\!\left(\omega_n t\right)
\end{equation}

where $\omega_n = \sqrt{1/(L_{dc}C_{dc}) - (R_{dc}/(2L_{dc}))^2}$ is the natural frequency of the LC circuit.

The SAMBP DC protection criterion:
\begin{equation}
\frac{di_{dc}}{dt}\bigg|_{t=0} = \frac{V_{dc,0}}{L_{dc}} > \left(\frac{di}{dt}\right)_{threshold}
\end{equation}

\section{Sub-Synchronous Resonance (SSCI) — TR-78}

SSCI between a DFIG and series-compensated line occurs when:

\begin{equation}
\label{eq:ssci}
f_{SSCI} = f_0 - f_r = f_0 - f_0\sqrt{\frac{X_C}{X_{total}}}
\end{equation}

The SSCI detector algorithm:

\begin{algorithm}[H]
\caption{SSCI Detector — TR-78}
\label{alg:ssci}
\KwIn{Current samples $i(t)$, sampling rate $f_s$}
\KwOut{SSCI alarm, frequency $f_{SSCI}$}
\BlankLine
Compute Prony decomposition of $i(t)$ over sliding window $W$\;
\For{each mode $m$ with frequency $f_m$}{
  \If{$f_m < f_0$ AND $f_m > 5$\,Hz AND growth rate $\sigma_m > 0$}{
    $f_{SSCI} \leftarrow f_m$\;
    Trigger SSCI alarm\;
    Issue damping command to DFIG RSC\;
  }
}
\end{algorithm}

\section{Ferroresonance Detection (TR-84)}

Ferroresonance produces chaotic, sustained overvoltages. SAMBP uses a 20-scenario library with Lyapunov exponent detection:

\begin{equation}
\label{eq:lyapunov}
\Lambda = \lim_{T\to\infty}\frac{1}{T}\ln\frac{\norm{\delta\bvec{x}(T)}}{\norm{\delta\bvec{x}(0)}} > 0 \implies \text{chaotic (ferroresonance)}
\end{equation}

\section{Cold Load Pickup (CLPU) — TR-85}

The CLPU current estimator:
\begin{equation}
\label{eq:clpu}
I_{CLPU}(t) = I_{steady}\left[1 + k_{cold}\cdot e^{-t/\tau_{CLPU}}\right]
\end{equation}

where $k_{cold} \in [2, 5]$ depends on load type and disconnection duration. The adaptive overcurrent threshold:
\begin{equation}
I_{pickup,adaptive}(t) = I_{CLPU}(t)\cdot(1 + \delta_{margin})
\end{equation}

\section{Geomagnetic Disturbance (TR-81)}

The GIC LP model from TR-81:
\begin{equation}
\label{eq:gic_lp}
\min_{\bvec{I}_{GIC}} \bvec{c}^\top\bvec{I}_{GIC} \quad \text{s.t.} \quad \bvec{A}\bvec{I}_{GIC} = \bvec{b}, \quad \bvec{I}_{GIC} \geq 0
\end{equation}

subject to network Kirchhoff constraints and transformer DC offset limits per IEC 60076-7.

  \item Load level: $P_{load} \in \{0.3, 0.7, 1.0\}$\,\pu{}
\end{itemize}

The nearest-neighbour scenario is retrieved in $O(\log N_{sc})$ via a KD-tree indexed on $[\rho_{IBR}, R_F, P_{load}]$.

\section{Computational Budget}

EKF update: $\sim$\SI{0.8}{\milli\second} per cycle on ARM Cortex-A72.\\
Trajectory generation: $\sim$\SI{3}{\milli\second} for 100-ms horizon.\\
Bayesian fusion: $\sim$\SI{0.2}{\milli\second}.\\
\textbf{Total}: $<$\SI{5}{\milli\second} $\ll$ \SI{50}{\milli\second} protection window. \cmark

\chapter{Cybersecurity — GOOSE Anomaly Detection (TR-82)}
\label{ch:cyber}
%%============================================================================

\section{Threat Model}

IEC 61850 GOOSE messages are broadcast over Ethernet (layer 2, multicast) with no authentication in base standard. Threat vectors include:

\begin{itemize}
  \item \textbf{Replay attack}: Attacker replays a captured GOOSE trip message.
  \item \textbf{Injection attack}: Attacker injects a spoofed GOOSE with stNum/sqNum manipulated.
  \item \textbf{DoS}: Flooding the Ethernet network to delay legitimate GOOSE delivery.
\end{itemize}

\section{Anomaly Detection Algorithm}

\begin{algorithm}[H]
\caption{GOOSE Anomaly Detector — TR-82}
\label{alg:goose}
\KwIn{GOOSE message stream $\{m_k\}$}
\KwOut{Anomaly flag, confidence score}
\BlankLine
\ForEach{message $m_k$}{
  Extract $(stNum_k, sqNum_k, t_k, DataSet_k)$\;
  \If{$stNum_k < stNum_{k-1}$ \textbf{OR} $sqNum_k \neq sqNum_{k-1}+1$}{
    Flag as sequence anomaly\;
  }
  $\Delta t_k \leftarrow t_k - t_{k-1}$\;
  \If{$\Delta t_k < T_{min}$ \textbf{OR} $\Delta t_k > T_{max,retransmit}$}{
    Flag as timing anomaly\;
  }
  Compute SPRT on $\Delta t_k$ under $\mathcal{H}_1$ (attack) vs $\mathcal{H}_0$ (normal)\;
  \If{$\Lambda_k \geq \ln(B)$}{
    Alert IEC 62351-6 security module\;
    Drop and log message $m_k$\;
  }
}
\end{algorithm}

\section{IEC 62351-6 Compliance}

TR-82 implements the authentication extensions per IEC 62351-6:
\begin{itemize}
  \item HMAC-SHA256 message authentication codes appended to GOOSE APDU.
  \item Key management via IEC 62351-8 role-based access control.
  \item Tested against 5 attack scenarios: replay, injection, DoS, MitM, rogue publisher.
\end{itemize}

Detection rates: replay 99.1\%, injection 97.8\%, DoS 94.2\%.

\section{TR-86: Multi-Terminal 87L}

Multi-terminal line differential requires generalising the two-terminal trip logic (Eq.~\ref{eq:87l_diff}) to $N$ terminals:

\begin{equation}
\label{eq:87l_mt}
I_{diff}^{MT} = \left|\sum_{k=1}^{N} \bvec{I}_k\right|, \quad
I_{rest}^{MT} = \sum_{k=1}^{N} |\bvec{I}_k|
\end{equation}

The challenge is synchronising $N$ phasor measurements with latency $\tau_k$ each. SAMBP will extend the EKF to estimate communication delay as an additional state variable.

\section{TR-90: Federated Learning}

Relays in different substations can share gradient updates without revealing raw fault data:

\begin{equation}
\bvec{w}^{(t+1)} = \sum_{k=1}^{K}\frac{n_k}{N}\bvec{w}_k^{(t)}
\end{equation}

where $\bvec{w}_k^{(t)}$ is relay $k$'s local model weight, $n_k$ is local data size.

\section{TR-92: Reinforcement Learning}

Formulate threshold adaptation as a Markov Decision Process:
\begin{itemize}
  \item \textbf{State}: $\bvec{s}_t = [\rho_{IBR}, R_F, P_{load}, \text{agree\_rate}_{t-1}]$
  \item \textbf{Action}: $\bvec{a}_t = [k_r, I_{min}, k_1, k_2]$ (relay parameters)
  \item \textbf{Reward}: $r_t = \text{agree\_rate}_t - \lambda_{maltrip}\cdot\mathbb{1}[\text{mal-trip}]$
\end{itemize}
